    # GenerationPlan owns review/commit lifecycle after Phase 4 Slice 8.
    # Canonical publication succeeds without synchronizing the legacy
    # planning_stages workflow status.
    refreshed = planning.get_session(session["id"])
    assert refreshed["stages"][0]["status"] == "ready"
